function fresh(){return {profile:{name:'',dept:''},items:JSON.parse(JSON.stringify(DEFAULT_ITEMS)),days:{},notes:{}};}
function normalize(d){ if(!d||!Array.isArray(d.items)) return fresh(); d.days=d.days||{}; d.notes=d.notes||{}; d.profile=d.profile||{name:'',dept:''}; return d; }

let docRef=null, saveTimer=null, saving=false, dirty=false, storMode='connecting';
function setStor(m){ storMode=m; $('stor').textContent = m==='cloud'?'ข้อมูลเก็บในบัญชีของคุณ (เห็นได้เฉพาะคุณ)' : m==='fail'?'บันทึกเข้าบัญชีไม่ได้ ตอนนี้เก็บในเบราว์เซอร์นี้เท่านั้น' : m==='local'?'ข้อมูลเก็บในเบราว์เซอร์นี้เท่านั้น ควรสำรองเป็นระยะ' : 'กำลังเชื่อมต่อ…'; }
function save(){
  try{ localStorage.setItem(KEY,JSON.stringify(db)); }catch(e){}
  if(docRef){ dirty=true; clearTimeout(saveTimer); saveTimer=setTimeout(flush,700); }
}
async function flush(){
  if(!docRef||!dirty) return;
  if(saving){ saveTimer=setTimeout(flush,300); return; }
  saving=true; dirty=false;
  try{ await docRef.set({data:db,savedAt:Date.now()}); setStor('cloud'); }
  catch(e){ setStor('fail'); }
  saving=false; if(dirty) flush();
}
async function connect(){
  try{
    if(!window.claude||!claude.use) throw 0;
    const [d,u]=await Promise.all([claude.use('db'),claude.use('user')]);
    if(!d||!u) throw 0;
    const id=await u.id(); if(!id) throw 0;
    const ref=d.doc('data/users/'+id+'/p4p');
    const snap=await ref.get();
    const remote=snap.exists&&snap.data()&&snap.data().data;
    docRef=ref;
    if(remote&&Array.isArray(remote.items)){ db=normalize(remote); try{localStorage.setItem(KEY,JSON.stringify(db));}catch(e){} setStor('cloud'); render(); }
    else { dirty=true; setStor('cloud'); await flush(); }
  }catch(e){ docRef=null; setStor('local'); }
}

function todayISO(){const d=new Date();return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0');}
function esc(s){return String(s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));}
function fmt(n){return (Math.round(n*100)/100).toLocaleString('th-TH',{maximumFractionDigits:2});}
function thDate(iso){const [y,m,d]=iso.split('-').map(Number);return new Date(y,m-1,d).toLocaleDateString('th-TH',{weekday:'long',day:'numeric',month:'long',year:'numeric'});}
function thDay(iso){const [y,m,d]=iso.split('-').map(Number);return new Date(y,m-1,d).toLocaleDateString('th-TH',{weekday:'short',day:'numeric',month:'short'});}
function thMonth(ym){const [y,m]=ym.split('-').map(Number);return new Date(y,m-1,1).toLocaleDateString('th-TH',{month:'long',year:'numeric'});}
function shiftDate(iso,n){const [y,m,d]=iso.split('-').map(Number);const t=new Date(y,m-1,d+n);return t.getFullYear()+'-'+String(t.getMonth()+1).padStart(2,'0')+'-'+String(t.getDate()).padStart(2,'0');}
function daysInMonth(ym){const [y,m]=ym.split('-').map(Number);return new Date(y,m,0).getDate();}

/* ---------- scoring ---------- */
function setQty(date,id,q){
  q=Math.max(0,Math.round((Number(q)||0)*100)/100);
  const it=db.items.find(x=>x.id===id); if(!it) return;
  db.days[date]=db.days[date]||{};
  if(q===0) delete db.days[date][id]; else db.days[date][id]={q,p:it.pts};
  if(!Object.keys(db.days[date]).length) delete db.days[date];
  save();
}
function dayTotal(date){const d=db.days[date]||{};let s=0;for(const k in d) s+=d[k].q*d[k].p;return s;}
function monthSummary(ym){
  const rows={},perDay={};let total=0,workDays=0;
  for(const date in db.days){
    if(!date.startsWith(ym)) continue;
    let dt=0;
    for(const id in db.days[date]){const e=db.days[date][id];const r=rows[id]=rows[id]||{id,qty:0,pts:0};r.qty+=e.q;r.pts+=e.q*e.p;dt+=e.q*e.p;}
    perDay[date]=dt;total+=dt;workDays++;
  }
  return {rows,total,perDay,workDays};
}
function itemName(id){const it=db.items.find(x=>x.id===id);return it?it.name:'(รายการที่ถูกลบ)';}

function csvText(){

}

async function offerFile(boxId,filename,text,fallbackTitle){
  try{
    const dl=window.claude&&claude.use?await claude.use('downloads'):null;

    await dl.save({filename,data:text});
    toast('บันทึกไฟล์แล้ว');
  }catch(e){
    if(e&&e.code==='declined') return;
    showText(boxId,fallbackTitle,text.replace(/^﻿/,''));
  }
}

function bindSet(){
  $('pname').onchange=()=>{db.profile.name=$('pname').value.trim();save();render();};
  $('pdept').onchange=()=>{db.profile.dept=$('pdept').value.trim();save();render();};
  document.querySelectorAll('#view .item[data-id]').forEach(el=>{
    const it=db.items.find(x=>x.id===el.dataset.id);
    el.querySelector('.iname').onchange=e=>{it.name=e.target.value.trim()||it.name;save();toast('บันทึกแล้ว');};
    el.querySelector('.ipts').onchange=e=>{it.pts=Number(e.target.value)||0;save();toast('บันทึกแล้ว');};
    el.querySelector('.itog').onclick=()=>{it.active=!it.active;save();render();};
    el.querySelector('.idel').onclick=()=>{
      const used=Object.values(db.days).some(d=>d[it.id]);
      const msg=used?'รายการ "'+it.name+'" มีข้อมูลที่บันทึกไว้แล้ว ถ้าลบ ชื่อรายการจะหายจากสรุป แนะนำให้กด "ซ่อน" แทน ต้องการลบจริงหรือไม่':'ลบรายการ "'+it.name+'" หรือไม่';
      ask(msg,'ลบ',()=>{db.items=db.items.filter(x=>x!==it);save();render();});
    };
  });
  $('add').onclick=()=>{db.items.push({id:'i'+Date.now().toString(36),name:'รายการใหม่',pts:1,active:true});save();render();};
  $('bk').onclick=()=>offerFile('outbox','p4p-backup-'+todayISO()+'.json',JSON.stringify(db),'ข้อมูลสำรอง (คัดลอกเก็บไว้ในที่ปลอดภัย)');
  $('rs').onclick=()=>$('file').click();
  $('file').onchange=()=>{
    const f=$('file').files[0]; if(!f) return;
    const r=new FileReader();
    r.onload=()=>{
      let d; try{ d=JSON.parse(r.result); if(!d.items||!d.days) throw 0; }catch(e){ toast('ไฟล์ไม่ถูกต้อง'); return; }
      ask('กู้คืนจะแทนที่ข้อมูลปัจจุบันทั้งหมด ดำเนินการต่อหรือไม่','กู้คืน',()=>{db=normalize(d);save();render();toast('กู้คืนแล้ว');});
    };
    r.readAsText(f);
  };
}
